\begin{frame}{Two independent reductions, one real turn}
    \large
    \begin{itemize}
        \item \texttt{enemy\_health = enemy\_health - player\_attack} then \texttt{player\_health = player\_health - enemy\_attack}
        \item Both fighters act in the same turn, each reduction fully independent of the other.
    \end{itemize}
\end{frame}


\begin{frame}{\texttt{and} requires BOTH fighters still alive}
    \large
    \begin{itemize}
        \item \texttt{while player\_health > 0 and enemy\_health > 0:}
        \item The battle only continues while NEITHER side has been defeated yet; either one reaching 0 ends it immediately.
    \end{itemize}
\end{frame}


\begin{frame}{BOSS — zero functions, a complete playable battle}
    \large
    \begin{itemize}
        \item Both fighters' health, a turn loop, and a final \texttt{if/else} announcing the winner once the loop ends
        \item A real, complete, playable encounter built from nothing but variables, loops, and if/else.
    \end{itemize}
\end{frame}

\begin{frame}{Real-World Examples}
    \large
    \begin{itemize}
        \item This is genuinely the same architecture behind the combat system of countless real shipped turn-based games
        \item Two independent health pools, a shared turn loop, and a clear win/lose message once it resolves.
    \end{itemize}
\end{frame}
