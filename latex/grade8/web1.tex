\begin{frame}{A real \texttt{<h1>} tag needs an opening AND closing half}
    \large
    \begin{itemize}
        \item \texttt{<h1>My Page</h1>}
        \item This is genuinely the first real HTML most students ever type, moving from a drag-and-drop "heading" block to the actual tag underneath it.
    \end{itemize}
\end{frame}


\begin{frame}{A page usually has exactly ONE heading}
    \large
    \begin{itemize}
        \item Just like Grade 7's Story Studio taught with blocks, a real page almost always has one \texttt{<h1>}
        \item The single thing that says what the whole page is about.
    \end{itemize}
\end{frame}


\begin{frame}{A \texttt{<p>} paragraph needs real, non-empty text}
    \large
    \begin{itemize}
        \item An empty tag technically exists but says nothing
        \item The checker genuinely verifies there is real text inside, not just that the tag is present.
    \end{itemize}
\end{frame}

\begin{frame}{Real-World Examples}
    \large
    \begin{itemize}
        \item Every real news homepage, blog post, or product page still starts with exactly these two tags
        \item A heading naming the page, a paragraph saying something real. Minecraft's recipe book pages have the same shape: one item name (heading), one description (paragraph).
    \end{itemize}
\end{frame}
