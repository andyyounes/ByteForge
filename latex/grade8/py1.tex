\begin{frame}{A variable is still a named box — just typed now}
    \large
    \begin{itemize}
        \item \texttt{name = "Alex"} is exactly Grade 7's "set" block, just written with an \texttt{=} sign instead of snapped together
        \item Same idea, real Python syntax.
    \end{itemize}
\end{frame}


\begin{frame}{Text needs quotes; numbers don't}
    \large
    \begin{itemize}
        \item \texttt{name = "Alex"} (text, in quotes) vs. \texttt{age = 13} (a number, no quotes)
        \item Mixing this up is a genuinely common first real Python bug.
    \end{itemize}
\end{frame}


\begin{frame}{BOSS — an f-string combines multiple variables in ONE print}
    \large
    \begin{itemize}
        \item \texttt{print(f"My name is \{name\}, I am \{age\}, and I like \{fav\_thing\}.")}
        \item An f-string starts with \texttt{f} before the quotes, and drops variables directly inside \texttt{\{\}} without needing separate print calls or \texttt{+} concatenation.
    \end{itemize}
\end{frame}

\begin{frame}{Real-World Examples}
    \large
    \begin{itemize}
        \item Every real sign-up form storing your name, age, and a preference as separate named fields is doing exactly this
        \item Three named boxes, combined into one readable message on a confirmation screen.
    \end{itemize}
\end{frame}
