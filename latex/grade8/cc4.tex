\begin{frame}{Two independent characters need two independent sets of variables}
    \large
    \begin{itemize}
        \item \texttt{player\_health}, \texttt{enemy\_health}, \texttt{enemy\_attack}
        \item Separate variables tracking separate things, the exact same "multiple independent pieces of state" idea from the Text Adventure's inventory, applied to combat.
    \end{itemize}
\end{frame}


\begin{frame}{The SAME subtraction pattern, a different attacker}
    \large
    \begin{itemize}
        \item \texttt{player\_health = player\_health - enemy\_attack}
        \item Identical shape to Player attacking Enemy, just with the roles and variable names swapped.
    \end{itemize}
\end{frame}

\begin{frame}{Real-World Examples}
    \large
    \begin{itemize}
        \item Every real multiplayer or PvE game genuinely tracks each character's stats as separate, independent variables
        \item One player's health dropping never touches another character's completely separate health variable.
    \end{itemize}
\end{frame}


\begin{frame}{Let Them Discover}
    \Large
    In any 2-character battle you know (a boss fight, a PvP match), name the FOUR separate numbers being tracked (each side's health and attack). Are any of them ever mixed up, or genuinely kept independent?
\end{frame}
