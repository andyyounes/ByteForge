\begin{frame}{if/elif/else branches a story on a real choice}
    \large
    \begin{itemize}
        \item \texttt{if choice == "left": ... elif choice == "right": ... else: ...}
        \item Three genuinely different paths from ONE variable, exactly Grade 7's branching-story idea in real Python syntax.
    \end{itemize}
\end{frame}


\begin{frame}{An inventory that GROWS as the story plays}
    \large
    \begin{itemize}
        \item \texttt{inventory = []} starts empty; each room branch calls \texttt{.append()} with that room's specific item
        \item The SAME list ends up holding whatever the player actually picked up, not a fixed pre-written set.
    \end{itemize}
\end{frame}


\begin{frame}{A while loop wraps the whole adventure}
    \large
    \begin{itemize}
        \item \texttt{while not game\_over:} keeps the story running turn after turn until a condition (\texttt{game\_over = True}, set after 3 turns) ends it
        \item The loop must change its own condition, same rule as every while loop before it.
    \end{itemize}
\end{frame}


\begin{frame}{BOSS — branching, a growing list, and a loop, together}
    \large
    \begin{itemize}
        \item A real playable text adventure needs genuinely all three: rooms that branch, an inventory that grows from real choices, and a turn loop controlling when it ends.
    \end{itemize}
\end{frame}

\begin{frame}{Real-World Examples}
    \large
    \begin{itemize}
        \item Every real text adventure (Zork and its descendants) and every branching dialogue system in a modern RPG is built from exactly these three ideas
        \item If/elif branching, a growing inventory list, and a loop controlling how long play continues.
    \end{itemize}
\end{frame}
