\begin{frame}{CSS decides HOW a page looks, not WHAT is on it}
    \large
    \begin{itemize}
        \item \texttt{body \{ background-color: \#FFF7E6; \}} and \texttt{h1 \{ color: \#2F3B52; \}} change appearance only
        \item The actual HTML content (heading, paragraph) is completely untouched by either rule.
    \end{itemize}
\end{frame}


\begin{frame}{An EXACT hex color, checked precisely}
    \large
    \begin{itemize}
        \item The checker compares the browser's own computed style
        \item \texttt{getComputedStyle} normalizes any hex color to an exact \texttt{rgb(...)} value, so "close enough" genuinely is not close enough; \texttt{\#FFF7E6} must become exactly \texttt{rgb(255, 247, 230)}.
    \end{itemize}
\end{frame}


\begin{frame}{A rule can target just ONE kind of element}
    \large
    \begin{itemize}
        \item \texttt{h1 \{ color: ... \}} only touches headings
        \item Paragraphs keep whatever color they already had, a targeted rule rather than a blanket one.
    \end{itemize}
\end{frame}

\begin{frame}{Real-World Examples}
    \large
    \begin{itemize}
        \item Every real website's consistent brand colors (the exact blue on every Facebook button, the exact red on every YouTube icon) are hex values set in CSS exactly like this
        \item Designers hand developers precise hex codes, not "kind of blue".
    \end{itemize}
\end{frame}
