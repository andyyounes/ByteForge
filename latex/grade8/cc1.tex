\begin{frame}{Health is just a variable that goes DOWN}
    \large
    \begin{itemize}
        \item \texttt{health = health - attack\_value}
        \item The exact same "read, subtract, store back" arithmetic from every earlier chapter, now genuinely meaning something inside a battle.
    \end{itemize}
\end{frame}


\begin{frame}{An attack is a fixed value, subtracted once per hit}
    \large
    \begin{itemize}
        \item \texttt{attack\_value = 15} is a plain number
        \item The same value subtracted from health however many times an attack lands, no special "combat" syntax needed.
    \end{itemize}
\end{frame}

\begin{frame}{Real-World Examples}
    \large
    \begin{itemize}
        \item Every RPG's damage calculation, at its core, really is exactly this line of arithmetic
        \item A health number decreasing by an attack number, dressed up with animations and sound.
    \end{itemize}
\end{frame}
