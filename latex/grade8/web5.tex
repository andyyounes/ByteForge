\begin{frame}{One CSS rule, applied everywhere at once}
    \large
    \begin{itemize}
        \item \texttt{body \{ background-color: \#FFF7E6; \}} and \texttt{h1 \{ color: \#2F3B52; \}} apply to the WHOLE page automatically
        \item However many sections exist, the same two rules cover all of them with zero repetition.
    \end{itemize}
\end{frame}


\begin{frame}{The FIRST real JavaScript idea: one click, one change}
    \large
    \begin{itemize}
        \item \texttt{document.querySelector("button").addEventListener("click", function() \{...\})}
        \item A real click handler that changes exactly one piece of text, the smallest possible genuine interactive behaviour.
    \end{itemize}
\end{frame}


\begin{frame}{The SECOND real JS idea: a variable the click updates}
    \large
    \begin{itemize}
        \item \texttt{let clicks = 0;} declared OUTSIDE the handler survives between clicks; inside the handler, \texttt{clicks = clicks + 1;} then displaying it is the exact same "read, change, store back" pattern from every variable chapter before it.
    \end{itemize}
\end{frame}
