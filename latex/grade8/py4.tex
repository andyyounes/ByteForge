\begin{frame}{\texttt{for} is for a KNOWN count; \texttt{while} is for UNKNOWN}
    \large
    \begin{itemize}
        \item A \texttt{for} loop is right when you know exactly how many times in advance; a \texttt{while} loop keeps going for as long as a condition stays true
        \item Right when you don't know the count ahead of time.
    \end{itemize}
\end{frame}


\begin{frame}{A while loop MUST change its own condition}
    \large
    \begin{itemize}
        \item \texttt{while count > 0:} paired with \texttt{count = count - 1} inside the loop
        \item Skip the decrement and the loop genuinely never stops, a real and common bug.
    \end{itemize}
\end{frame}


\begin{frame}{BOSS — the same shape, a bigger step}
    \large
    \begin{itemize}
        \item \texttt{health = health - 25} each pass, \texttt{while health > 0:} as the condition
        \item Identical structure to the countdown, just counting down by a different amount toward a real "Game Over" message.
    \end{itemize}
\end{frame}

\begin{frame}{Real-World Examples}
    \large
    \begin{itemize}
        \item A real video game health bar counting down by fixed damage amounts, or a countdown timer before a rocket launch, are both genuinely \texttt{while}-loop shaped
        \item Repeat until a condition (health, or the countdown itself) changes to false.
    \end{itemize}
\end{frame}
