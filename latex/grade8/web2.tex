\begin{frame}{A link needs an \texttt{href} — a real destination}
    \large
    \begin{itemize}
        \item \texttt{<a href="https://example.com">Click here</a>}
        \item A link missing its href is like a signpost with an arrow but no name on it, technically still a tag but useless to a real reader.
    \end{itemize}
\end{frame}


\begin{frame}{A link goes elsewhere; a button acts here}
    \large
    \begin{itemize}
        \item This is the exact same distinction Grade 7's Story Studio 1 taught with blocks
        \item Now typed as real, separate tags, \texttt{<a>} vs \texttt{<button>}.
    \end{itemize}
\end{frame}


\begin{frame}{A button on its own does genuinely nothing yet}
    \large
    \begin{itemize}
        \item \texttt{<button>} with no JavaScript attached simply sits there when clicked
        \item That is expected and correct this week; real behaviour attaches to it later (Web Story project, Session 5-6).
    \end{itemize}
\end{frame}

\begin{frame}{Real-World Examples}
    \large
    \begin{itemize}
        \item Every "read more" link and "subscribe" button on a real news site is exactly these two tags, doing exactly these two different jobs, at real scale.
    \end{itemize}
\end{frame}
