\begin{frame}{A battle is a while loop wearing combat clothes}
    \large
    \begin{itemize}
        \item \texttt{while health > 0:} is the exact same "unknown count, until a condition changes" idea from Semester 2's while loops
        \item Now controlling how many turns a fight lasts.
    \end{itemize}
\end{frame}


\begin{frame}{The loop must still change its own condition}
    \large
    \begin{itemize}
        \item \texttt{health = health - 10} inside the loop is what eventually makes \texttt{health > 0} false
        \item Skip it, and the battle genuinely never ends.
    \end{itemize}
\end{frame}

\begin{frame}{Real-World Examples}
    \large
    \begin{itemize}
        \item Every turn-based RPG battle (Pokémon, Final Fantasy) runs on exactly this loop shape underneath
        \item Repeat combat turns while both sides remain above 0 health.
    \end{itemize}
\end{frame}
