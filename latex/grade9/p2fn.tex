\begin{frame}{Why a function, not just inline code}
    \large
    \begin{itemize}
        \item With 2+ enemies, "which one do I attack" is a real decision worth naming
        \item \texttt{def pick\_target():} isolates that ONE decision cleanly
    \end{itemize}

\begin{frame}{Decomposition --- a genuinely important real skill}
    \large
    \begin{itemize}
        \item Break one tangled problem into a named, callable piece that does ONE job
        \item Every mission from here on REQUIRES this, not just allows it
    \end{itemize}
\end{frame}


\begin{frame}{The "find the minimum" scanning pattern}
    \large
    \begin{itemize}
        \item Assume the first item is the answer
        \item Check every OTHER item --- update only if something beats it
        \item This exact pattern finds ANY extreme value, min or max, in ANY list
    \end{itemize}
\end{frame}
