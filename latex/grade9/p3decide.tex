\begin{frame}{Minimum uses the same pattern, flipped}
    \large
    \begin{itemize}
        \item "Cheapest route" is a minimum search --- same scanning idea as maximum, just \texttt{<} instead of \texttt{>}
        \item Track the NAME and the cost together as you scan
    \end{itemize}
\end{frame}


\begin{frame}{A running counter driven by an IF inside a loop}
    \large
    \begin{itemize}
        \item "How many contacts have more than 50 hp" --- loop + IF + counter
        \item Counting is NOT the same as summing --- the value itself never gets added anywhere
    \end{itemize}
\end{frame}


\begin{frame}{A running total that goes DOWN, not up}
    \large
    \begin{itemize}
        \item Start a budget, SUBTRACT each cost as you loop --- same shape as adding, opposite direction
    \end{itemize}
\end{frame}
