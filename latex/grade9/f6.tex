\begin{frame}{Each flowchart shape has ONE specific job}
    \large
    \begin{itemize}
        \item Oval (terminator) --- marks START or END, exactly one of each per flowchart
        \item Rectangle (process) --- a single action, e.g. "add 1 to total"
        \item Diamond (decision) --- a yes/no question with TWO arrows out
        \item Parallelogram (input/output) --- data going in, or being shown to the user
    \end{itemize}
\end{frame}


\begin{frame}{A decision diamond becomes an IF/ELSE in real code}
    \large
    \begin{itemize}
        \item The diamond IS the condition
        \item Each arrow out of it IS one branch (the IF, and the ELSE)
        \item Plan the diagram first, THEN turn it into real code --- much easier both steps
    \end{itemize}
\end{frame}

\begin{frame}{Real-World Examples}
    \large
    \begin{itemize}
        \item Every food delivery app's "order status" screen (placed -> preparing -> on the way -> delivered) is really a flowchart brought to life, one box per status.
        \item A vending machine is a real flowchart: insert money (input) -> enough money? (decision) -> dispense (process) or ask for more (loop back).
    \end{itemize}
\end{frame}
