\begin{frame}{for i in range(4): --- say it once, it happens 4 times}
    \large
    \begin{itemize}
        \item A square is the SAME 2 actions repeated 4 times
        \item Write the pattern ONCE inside the loop instead of 4 separate times
        \item Stars now count LINES of code, not moves --- fewer lines, same result
    \end{itemize}
\end{frame}


\begin{frame}{The loop variable is real, usable data}
    \large
    \begin{itemize}
        \item \texttt{for i in range(1, 7):} makes i genuinely count 1, 2, 3... 6
        \item \texttt{forward(40 * i)} makes each leg automatically longer --- i is DATA, not just a counter
        \item Change "6 legs" to "12 legs" and only ONE number needs changing
    \end{itemize}
\end{frame}


\begin{frame}{360 / sides = the turn angle, for ANY regular shape}
    \large
    \begin{itemize}
        \item A hexagon turns 60 degrees each corner (360/6)
        \item The exact same formula works for a triangle, square, or octagon
    \end{itemize}
\end{frame}

\begin{frame}{Real-World Examples}
    \large
    \begin{itemize}
        \item A washing machine's spin cycle is a real loop: repeat "spin, pause" a set number of times, exactly like \texttt{for i in range(N):}.
        \item Minecraft building a straight fence or row of torches by hand is literally what a loop automates --- the same repeated action, over and over.
    \end{itemize}
\end{frame}
