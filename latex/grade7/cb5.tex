\begin{frame}{A vague prompt gets a vague, generic result}
    \large
    \begin{itemize}
        \item "get help writing about dogs" gives an AI almost nothing to work with
        \item No length, no tone, no audience, no format
        \item So the result is whatever the AI guesses you might want.
    \end{itemize}
\end{frame}


\begin{frame}{A specific prompt has exactly 3 real parts}
    \large
    \begin{itemize}
        \item A subject (what you want), a constraint (length, tone, or audience), and a format (list, paragraph, or steps)
        \item "Write a funny 3-sentence paragraph about why dogs are better than cats, for a 12-year-old" has genuinely all three.
    \end{itemize}
\end{frame}


\begin{frame}{"Make me a story" is missing all three parts at once}
    \large
    \begin{itemize}
        \item No subject (a story about WHAT), no constraint (how long, what tone), no format (a poem? a paragraph? a list of scenes?)
        \item Identifying exactly what's missing is the real skill, not just noticing it's vague.
    \end{itemize}
\end{frame}

\begin{frame}{Real-World Examples}
    \large
    \begin{itemize}
        \item Every professional who works with AI tools daily writes prompts with exactly these three parts, deliberately
        \item The difference between a vague prompt and a specific one is genuinely the difference between a useless first draft and a usable one.
    \end{itemize}
\end{frame}


\begin{frame}{Let Them Discover}
