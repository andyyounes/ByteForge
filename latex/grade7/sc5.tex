\begin{frame}{A rule that only fires past a threshold}
    \large
    \begin{itemize}
        \item There's no special "edge" concept in code
        \item \texttt{x > 400} IS the edge. Any boundary in a real game is built from exactly this.
    \end{itemize}
\end{frame}


\begin{frame}{IF can replace what's shown, not just add to it}
    \large
    \begin{itemize}
        \item "You Win!" and the running score are in different branches of the same if/else
        \item Only one is ever drawn on a given frame.
    \end{itemize}

    \begin{center}

\begin{frame}{Real-World Examples}
    \large
    \begin{itemize}
        \item A car's parking sensor beeping only once you're within a set distance of a wall
        \item The "beep" only fires past a threshold, exactly like this chapter's edge check. In Minecraft, fall damage only triggers past a height threshold
        \item No damage for a 1-block drop, real damage past \textasciitilde{}3 blocks
        \item The exact same "past a threshold" rule.
    \end{itemize}
\end{frame}
