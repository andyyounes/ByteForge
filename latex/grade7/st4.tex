\begin{frame}{A variable is a box with a name}
    \Large
    A named box holding a value your code can reuse, combine, or change later — the same code works no matter what's stored inside.
\end{frame}


\begin{frame}{Multiple variables, remembered together}
    \Large
    A real interactive story remembers MANY details — which item they picked, which room they're in, how much health they have.
\end{frame}


\begin{frame}{A variable is a box that can CHANGE}
    \Large
    \texttt{health = health - 20} reads the current value, subtracts 20, then stores the result back — that's what makes a variable useful for a story.
\end{frame}


\begin{frame}{BOSS — a page that remembers}
    \Large
    HTML gave structure, CSS gave style, onclick gave behaviour — variables give the page MEMORY.
\end{frame}

\begin{frame}{Real-World Example}
    \Large
    Your shopping cart total, a video's "watched 12:34 of 20:00" progress, a game's remembered high score between sessions — every one of these is a variable persisting a value.
\end{frame}
