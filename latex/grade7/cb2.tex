\begin{frame}{A file extension tells you what KIND of file it is}
    \large
    \begin{itemize}
        \item \texttt{.jpg}/\texttt{.png} are images, \texttt{.mp3}/\texttt{.wav} are audio, \texttt{.docx} is a document, \texttt{.mp4} is video, \texttt{.exe} is a program you run
        \item The three letters after the dot are a real, meaningful signal, not decoration.
    \end{itemize}
\end{frame}


\begin{frame}{KB, MB, GB — each about 1000x the last}
    \large
    \begin{itemize}
        \item A kilobyte, a megabyte, a gigabyte
        \item Each unit is roughly a thousand times bigger than the one before it, the same scaling idea used for every real file size you've ever seen on a device.
    \end{itemize}
\end{frame}


\begin{frame}{A folder is a container, not a file}
    \large
    \begin{itemize}
        \item A folder groups files together but isn't itself a document, image, or program
        \item A genuinely different kind of thing on a real file system.
    \end{itemize}
\end{frame}

\begin{frame}{Real-World Examples}
    \large
    \begin{itemize}
        \item Every real device you own sorts files by extension behind the scenes
        \item Tapping a \texttt{.jpg} opens a photo viewer, tapping a \texttt{.mp4} opens a video player, because the extension genuinely tells the operating system what kind of file it is.
    \end{itemize}
\end{frame}
